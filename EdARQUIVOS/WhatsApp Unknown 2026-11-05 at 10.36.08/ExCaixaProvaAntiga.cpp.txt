#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>


struct TpItem{
	char nome[50];
	float preco;
	TpItem *prox;
};

struct TpCliente{
	char nome[50];
	TpItem *itensComprados;
	float totalGasto;
	TpCliente *prox;
};

struct TpCaixa{
	char nomeAtendente[50];
	TpCliente *clientesAtendidos;
	TpCaixa *prox;
	TpCaixa *ant;
};

TpCaixa *NC_Caixa(char nomeAtendente[]){
	TpCaixa *nc = new TpCaixa;
	strcpy(nc->nomeAtendente, nomeAtendente);
	nc->clientesAtendidos = NULL;
	nc->prox = NULL;
	nc->ant = NULL;
}

TpCliente *NC_Cliente(char nome[]){
	TpCliente* nc = new TpCliente;
	strcpy(nc->nome, nome);
	nc->itensComprados = NULL;
	nc->prox = NULL;
	nc->totalGasto = 0;
}

TpItem *NC_Item(char nome[],float preco){
	TpItem *nc = new TpItem;
	strcpy(nc->nome,nome);
	nc->prox = NULL;
	nc->preco = preco;
}

TpCaixa *abrirCaixa(TpCaixa *caixas, char nomeAtendente[]){
	TpCaixa *novoCaixa = NC_Caixa(nomeAtendente);
	
	if(caixas == NULL){
		caixas = novoCaixa;
	}else{
		TpCaixa *atual = caixas;
		
		while(atual != NULL && strcmp(atual->nomeAtendente, nomeAtendente) < 0)
			atual = atual->prox;
		
		if(atual != NULL){
			if(atual->ant == NULL){
				novoCaixa->prox = caixas;
				caixas->ant = novoCaixa;
				caixas = novoCaixa;
			}else{
				novoCaixa->prox = atual;
				novoCaixa->ant = atual->ant;
				if(atual->prox != NULL){
					atual->prox->ant = novoCaixa;
				}
				atual = novoCaixa;
			}
		}
	}
	return caixas;
}

TpCaixa *buscarCaixaPorAtendente(TpCaixa *caixas, char nomeAtendente[]){
	TpCaixa *atual = caixas;
	while(atual != NULL && strcmp(atual->nomeAtendente, nomeAtendente) != 0)
		atual = atual->prox;
	return atual;
}

TpCaixa *adicionarCliente(TpCaixa *caixa, char nomeCliente[]){
	TpCliente *novoCliente = NC_Cliente(nomeCliente);
	
	if(caixa->clientesAtendidos == NULL){
		caixa->clientesAtendidos = novoCliente;
	}else{
		TpCliente *atual = caixa->clientesAtendidos;
		while(atual->prox != NULL)
			atual = atual->prox;
		atual->prox = novoCliente;
	}
	return caixa;
}


TpCliente *adicionarItem(TpCliente *cliente, char nomeItem[], float preco){
	TpItem *novoItem = NC_Item(nomeItem, preco);
	
	if(cliente->itensComprados == NULL){
		cliente->itensComprados = novoItem;
	}else{
		TpItem *atual = cliente->itensComprados;
		while(atual->prox != NULL)
			atual = atual->prox;
		atual->prox = novoItem;
	}
	return cliente;
}



float calcularTotal(TpItem *item){
	if(item == NULL)
		return 0;
	else
		return item->preco + calcularTotal(item->prox);
}

void liberarItens(TpItem *item){
	if(item != NULL){
		liberarItens(item->prox);
		delete(item);
	}
}
